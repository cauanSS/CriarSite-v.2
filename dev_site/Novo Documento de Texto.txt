3. assets/js/drag-drop.js (Drag & Drop)
JavaScriptdocument.addEventListener('DOMContentLoaded', () => {
  const blocks = document.querySelectorAll('.block');
  const canvas = document.getElementById('canvas');

  blocks.forEach(block => {
    block.addEventListener('dragstart', e => {
      e.dataTransfer.setData('text/plain', block.dataset.type);
    });
  });

  canvas.addEventListener('dragover', e => e.preventDefault());
  canvas.addEventListener('drop', e => {
    e.preventDefault();
    const type = e.dataTransfer.getData('text/plain');
    const newBlock = document.createElement('div');
    newBlock.className = 'dropped-block';
    newBlock.innerHTML = `<h2>${type.toUpperCase()}</h2><p>Conteúdo editável</p>`;
    newBlock.contentEditable = true;
    canvas.appendChild(newBlock);
  });
});
